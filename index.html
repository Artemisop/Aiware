function createPost() { let title = document.getElementById('postTitle').value.trim(); let content = document.getElementById('postContent').value.trim(); let category = document.getElementById('postCategory').value; if (!title || !content) { alert('Пожалуйста, заполните заголовок и содержание!'); return; } let posts = JSON.parse(localStorage.getItem('aiware_posts')); posts.unshift({ id: Date.now(), title: title, content: content, category: category, author: currentUser }); localStorage.setItem('aiware_posts', JSON.stringify(posts)); document.getElementById('postTitle').value = ''; document.getElementById('postContent').value = ''; alert('Публикация успешно создана!'); updateRealtimeData(); } function updateRealtimeData() { let users = JSON.parse(localStorage.getItem('aiware_users')) || []; let posts = JSON.parse(localStorage.getItem('aiware_posts')) || []; if (currentUser) { let activeUser = users.find(u => u.username === currentUser); if (activeUser) { activeUser.lastActive = Date.now(); localStorage.setItem('aiware_users', JSON.stringify(users)); } } let now = Date.now(); let onlineList = users.filter(u => u.lastActive && (now - u.lastActive) < 180000); document.getElementById('onlineUsers').innerText = onlineList.length > 0 ? onlineList.map(u => u.username).join(', ') : 'Нет пользователей онлайн'; document.getElementById('onlineCounter').innerText = `Всего: ${onlineList.length} (участников: ${onlineList.length}, гостей: 0)`; let newsCount = posts.filter(p => p.category === 'news').length; let updateCount = posts.filter(p => p.category === 'update').length; document.getElementById('countNewsThreads').innerText = newsCount; document.getElementById('countNewsMsgs').innerText = newsCount > 0 ? newsCount * 3 : 0; document.getElementById('countUpdateThreads').innerText = updateCount; document.getElementById('countUpdateMsgs').innerText = updateCount > 0 ? updateCount * 3 : 0; document.getElementById('statThreads').innerText = posts.length; document.getElementById('statMessages').innerText = posts.length > 0 ? posts.length * 3: 0; document.getElementById('statMembers').innerText = users.length; let latestUser = users[users.length - 1]; document.getElementById('statLatest').innerText = latestUser ? latestUser.username : '-'; } checkAuth(); updateRealtimeData(); renderChatMessages(); setInterval(updateRealtimeData, 10000);
